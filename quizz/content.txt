סִימָן סג
(ב) מְלֶאכֶת חוֹרֵשׁ
מַהוּתָהּ: הַרְפָּיַת גּוּשִׁי אֲדָמָה, וְיִשּׁוּר פְּנֵי הַקַרְקַע כְּדֵי לְהַכְשִׁירוֹ לִזְרִיעָה.
שְׁמוּשָׁה בַּמִשְׁכָּן: חָרְשׁוּ אֶת הַקַרְקַע כְּדֵי לִזְרֹעַ סַמְמָנִים לִצְבִיעַת הַיְרִיעוֹת.
א. הַחוֹרֵשׁ אוֹ הַחוֹפֵר אוֹ הָעוֹשֶׂה חָרִיץ בְּשַׁבָּת בְּקַרְקַע הָרְאוּיָה לִזְרִיעָה אוֹ לִנְטִיעָה, הֲרֵי זֶה אָסוּר מִן הַתּוֹרָה מִשּׁוּם מְלֶאכֶת חוֹרֵשׁ, שֶׁהֲרֵי מַרְפֶּה אֶת הַקַרְקַע וּמְתַקְנָהּ לִזְרִיעָה אוֹ לִנְטִיעָה.
ב. אָסוּר לְהַשְׁווֹת פְּנֵי קַרְקַע שֶׁל שָׁדֶה בְּשַׁבָּת, כְּגוֹן שֶׁהָיוּ בָהּ גַּבְשׁוּשִׁיּוֹת אוֹ עֲרֵמוֹת עָפָר וְרִדְּדָן, אוֹ שֶׁהָיוּ בָהּ גְּמּוֹת וּמִלְאָן בֶּעָפָר. שֶׁהַשְׁוְאֵת פְּנֵי הַקַרְקַע מְתַקֶנֶת וּמַכְשִׁירָה אֶת הַקַרְקַע לִזְרִיעָה טוֹבָה יוֹתֵר.
ג. הַחוֹפֵר גָּמָא בְּשָׂדֶה כְּדֵי לְהִשְׁתַּמֵשׁ בֶּעָפָר שֶׁמּוֹצִיא, וְלֹא עַל מְנָת לִזְרֹעַ שָׁם אָסוּר מִדִּבְרֵי סוֹפְרִים.
ד. הַמְנַפֶשׁ בְּעִקְרֵי הָאִילָנוֹת וְהַמְכַרְסֵם עֲשָׂבִים, וְכֵן הַמְזַבֵּל אֶת הַקַרְקַע הֲרֵי זֶה אָסוּר מִשּׁוּם חוֹרֵשׁ. שֶׁכָּל־פְּעָלָה שֶׁמְּיַפָּה וּמְתַקֶנֶת הַקַרְקַע לִזְרִיעָה אוֹ לִנְטִיעָה, אֲסוּרָה מִשּׁוּם חוֹרֵשׁ.
ה. אָסוּר לְכַבֵּד בְּמַטְאֲטֵא קַרְקַע שֶׁאֵינָהּ מְרְצֶפֶת בְּשַׁבָּת, שֶׁמֶא יַשְׁוֶה גְמּוֹת. אֲבָל קַרְקַע מְרְצֶפֶת – מְתָּר.
ו. מְתָּר לִגְדֹר עַל הָאָרֶץ מִטָה אוֹ כַּסֵּא אוֹ סַפְסָל, בֵּין גְּדוֹלִים בֵּין קְטַנִּים, וּבִלְבַד שֶׁלֹא יִתְכַּוּן לַעֲשׂוֹת חָרִיץ. וְכֵן מְתָּר לָצֵאת לְמָקוֹם הַמְעֹרֶב כַּדִּין עִם עֲגָלָה שֶׁל תִּינוֹק אַף שֶׁעוֹבֵר בְּדֶרֶךְ עָפָר, וְאֵין לָחוּשׁ לַעֲשִׂיַּת חָרִיץ.
ז. נִדְבָּק בְּמִנְעָלָיו טִיט אוֹ בּוֹץ, לֹא יְקַנְחֵם בַּקַרְקַע שֶׁאֵינָהּ מְרְצֶפֶת, שֶׁמָּא יָבֹא לְהַשְׁווֹת גְּמּוֹת, אֶלָּא יְקַנְחֵם בְּכֹתֶל אוֹ בְּחִדּוּדֵי הַמַּדְרֵגוֹת.
ח. אָסוּר לִשְׁטף אֶת הָרְצְפָה בְּשַׁבָּת, עַל־יְדֵי מַיִם אוֹ כָּל־חֹמֶר הֶעָשׂוּי לְנִקְוּי. וּמִכָּל־מְקוֹם, אִם נִתְלַכְלְכָה הָרִצְפָּה, מְתָּר לִשְׁפֹּךְ מַיִם רַק עַל הַמָּקוֹם הַמְלִכְלָךְ וּלְגְרְפָם עַל־יְדֵי מַנָּב גּוּמִי שֶׁאֵינוֹ סוֹפֵג.

סִימָן סד
(ג) מְלֶאכֶת קוֹצֵר
מַהוּתָהּ: תְּלִישַׁת צוֹמֵחַ מִמְקוֹר יְנִיקָתוֹ.
שׁמוּשָׁה בַּמַשְׁכְן: קָצְרוּ סַמְמָנִים כְּדֵי לַעֲשׂוֹת מֵהֶם צֶבַע.
א. הַקוֹצֵר תְּבוּאָה אוֹ קִטְנִית, וְהַבּוֹצֵר עֲנָבִים, וְהַגּוֹדֵר תְּמָרִים, וְהַמּוֹסֵק זֵיתִים וְכַיּוֹצֵא בָהֶם, הֲרֵי זֶה אָסוּר מִן הַתּוֹרָה מִשּׁוּם מְלֶאכֶת קוֹצֵר.
ב. הַתּוֹלֵשׁ בְּיָדוֹ דְּבָר מִמְקוֹם גִּדּוּלוֹ, וְכֵן הַמְנַעְנֵעַ אֶת הָאִילָן וּמַרְעִידוֹ כְּדֵי שֶׁיֵּשִׁיר פּרוֹתָיו, הֲרֵי זֶה אָסוּר מִשּׁוּם קוֹצֵר.
ג. אָסוּר לִתְלֹשׁ בְּשַׁבָּת אֲפִלּוּ מֵעָצִיץ שֶׁאֵינוֹ נָקוּב בְּתַחְתִּיתוֹ (כִּי כַּאֲשֶׁר יֵשׁ בּוֹ נֶקֶב בְּוַדַּאי אָסוּר, מִפְּנֵי שֶׁיְנִיקָתוֹ מַמָּשׁ מֵהַקַרְקַע).
ד. אָסוּר לְטַל עָצִיץ מֵעַל גַּבֵּי קַרְקַע וּלְהַנִּיחוֹ עַל גַּבֵּי יְתֵדוֹת, וַאֲפִלּוּ בְּעָצִיץ שֶׁאֵינוֹ נָקוּב, שֶׁבִּפְעָלָה זוֹ הֲרֵי הוּא מְמַעֵט אֶת יְנִיקָתוֹ מִן הַקַרְקַע. וְאִם הָיְתָה הַקַרְקַע מְרְצֶפֶת מְתָּר אֲפִלּוּ אִם הֶעָצִיץ נָקוּב.
ה. עֲנַף עֵץ פַּרוֹת תָּלוּשׁ מֵעֶרֶב שַׁבָּת, מְתָּר לְלַקֵט מִמֶּנּוּ אֶת הַפַּרוֹת בְּשַׁבָּת. וְכֵן מְתָּר לִקְטם בְּשַׁבָּת בְּשָׂמִים וּפְרָחִים מֵעֲנָפִים תְּלוּשִׁים אֲפִלּוּ כְּדֵי לְחַלְקָם לַאֲחֵרִים. וְדַוְקָא בְּיָדוֹ. אֲבָל בְּסַכִּין אָסוּר וְרְאֵה הֶעָרָה].
ו. הֲדַס הַמְחְבָּר לַקַרְקַע מְתָּר לִתְפַּס בּוֹ בְּיָדָיו וּלְבָרֵךְ עָלָיו וּלְהָרִיחַ בּוֹ בְּשַׁבָּת, וּבִלְבַד שֶׁלֹא יִתְלֹשׁ מִמֶּנּוּ. אֲבָל אֶתְרוֹג וְתַפּוּחַ וּשְׁאָר פּרוֹת הָרְאוּיִים לַאֲכִילָה, אָסוּר לְהָרִיחַ בָּהֶם בְּעוֹדָם מְחְבָּרִים לַקַרְקַע, גְּזֵרָה שֶׁמָּא יִתְלְשֵׁם כְּדֵי לְאָכְלָם.
ז. אָסוּר לַעֲלוֹת בְּאִילָן בְּשַׁבָּת, בֵּין לַח בֵּין יָבֵשׁ, וְאָסוּר לְהִתָּלוֹת בּוֹ אוֹ לְהִשָּׁעֵן עָלָיו בְּכָל־כֹּחוֹ אוֹ לְהִשְׁתַּמֵשׁ בּוֹ, כְּגוֹן לְהַנִּיחַ עָלָיו אֵיזֶה חֵפֶץ אוֹ לְהוֹרִיד מִמֶּנּוּ אֵיזֶה חֵפֶץ שֶׁהָיָה מְנָּח עָלָיו, גְּזֵרָה שֶׁמָּא יַעֲלֶה בָּאִילָן וְיִתְלֹשׁ מִמֶּנּוּ פַּרוֹת אוֹ עָלִים וַעֲנָפִים.
ה. אֵין רוֹכְבִין עַל גַּבֵּי בְהֵמָה בְּשַׁבָּת, וְאֵין נִתְלִים עָלֶיהָ, וְאֵין לְהִשְׁתַּמֵשׁ אֲפִלּוּ בְּצִדָּהּ, שֶׁמָּא יִתְלֹשׁ זְמוֹרָה כְּדֵי לְהַנְהִינָהּ וְיַעֲבֹר עַל קוֹצֵר. וְאָסוּר זֶה גַּם בְּמָקוֹם שֶׁאֵין זְמוֹרָה לַחְתּךְ, כְּגוֹן בַּמִּדְבָּר וְכַדּוֹמֶה.
ט. מְתָּר לִילֵךְ עַל גַּבֵּי עֲשָׂבִים בְּשַׁבָּת, בֵּין לַחִים בֵּין יְבֵשִׁים, לַמְרוֹת שֶׁהָעֲשָׂבִים נִתְלָשִׁים מִכֹּחַ דְּרִיסָתוֹ עֲלֵיהֶם, שֶׁהֲרֵי אֵינוֹ מִתְכַּוּן לִתְלֹשׁ. וְכֵן מְתָּר לָשֶׁבֶת אוֹ לִשְׁכַּב עַל הַדֶּשֶׁא בְּשַׁבָּת אוֹ לִפְרֹס שְׂמִיכָה וְלָשֶׁבֶת עָלֶיהָ, וּבִלְבַד שֶׁיִּזְהֵר שֶׁלֹא יִתְלֹשׁ בַּיָּדַיִם מִן הַמְחְבָּר.
וּ. אָסוּר לִרְדּוֹת חַלּוֹת דְּבַשׁ מִן הַכַּדֶּרֶת, מִפְּנֵי שֶׁדּוֹמֶה לְתוֹלֵשׁ.

סִימָן סה
(ד) מְלֶאכֶת מְעַמֵר
מַהוּתָהּ: אִסוּף גִּדּוּלֵי קַרְקַע תְּלוּשִׁים, בִּמְקוֹם גִּדּוּלָם, לַעֲרֵמָה.
שִׁמּוּשָׁהּ בַּמִּשְׁכָּן: אָסְפוּ אֶת הַפַּמְמָנִים.
א. הָאוֹסֵף שְׁבֶּלִים קְצוּרִים הַמְפָזְרִים מִמְקוֹם גִדּוּלָם וְעָשָׂה מֵהֶם עֲמָרִים (עֲרֵמוֹת), וְכֵן הָאוֹסֵף פַּרוֹת לְאַחַר תְּלִישָׁתָם מֵהָאִילָן, הַמְפְזְרִים סְבִיבוֹת הָאִילָן, וַאֲסָפָם יַחַד לְתוֹךְ סַל אוֹ קְפָּה וְכַיּוֹצֵא בָזֶה אָסוּר מִן הַתּוֹרָה מִשּׁוּם מְעַמָּר.
ב. הַמְעַמֵר דְּבָרִים שֶׁאֵינָם גְדֵלִים בַּקַרְקַע, כְּגוֹן הָאוֹסֵף מֶלַח מִמָּקוֹם שֶׁנּוֹצָר, וְכֵן הָאוֹסֵף בֵּיצִים מִלּוּל תַּרְנְגוֹלוֹת הַמְטִילוֹת - אָסוּר מִדִּבְרֵי סוֹפְרִים, מִשּׁוּם שֶׁדּוֹמֶה לִמְעַמֶּר.
ג. פַּרוֹת שֶׁנִּתְפַּזְרוּ בְּחָצֵר, אֶחָד הֵנָּה וְאֶחָד הֵנָּה, אֶסְרוּ חֲכָמִים לְאָסְפָם וְלִתְּנָם לְתוֹךְ סַל מִשּׁוּם עוֹבְדִין דְּחֹל (מַעֲשֵׂה חל), אֶלָּא אוֹסֵף מְעַט מְעַט וְאוֹכֵל. וְאִם נָפְלוּ בְּמָקוֹם אֶחָד, אַף שֶׁמְפְזְרִים מְתָּר לָתֵת לְתוֹךְ סַל אוֹ קְפָּה, שֶׁאֵין אֹרַח גָּדוֹל בַּאֲסִיפָתָם. אֶמְנָם אִם נִתְפַּזְרוּ הַפַּרוֹת בְּבַיִת, אֲפִלּוּ בְּכַמָּה מְקוֹמוֹת, מְתָּר לְקַבְּצָם וְלִתְּנָם לְתוֹךְ סַל אוֹ קְפָּה.
ד. הַמִּנְהָג לִזְרֹק בָּטְנִים וּשְׁקֵדִים עַל חֲתָנִים בְּשַׁבָּת וְאוֹסְפִים אוֹתָם הַיְלָדִים לְשַׁקִית, אֵין בָּזֶה אִסּוּר עוֹבָדִין דְּחל, הוֹאִיל וְעוֹשִׂים כֵּן לְשִׂמְחַת חָתָן, וְכָל־שֶׁכֵּן אִם זוֹרְקִים שְׁקֵדִים הַמְחָפִים בְּסְכָּרִיּוֹת שֶׁאֵין לֶאֱסר.
ה. הַמְקַבֵּץ דְבֵלָה וְעָשָׂה מִמֶּנָּה עִגּוּל, אוֹ שֶׁנְקֵב בִּתְאֵנִים וְהִכְנִיס הַחוּט בָּהֶן עַד שֶׁנִּתְקַבְּצוּ לְגוּשׁ אֶחָד – אָסוּר מִשּׁוּם מְעַמֶּר.
ו. חֲרוּזֵי פְּנִינִים שֶׁנִּפְרְדוּ מֵהַחוּט וְנִתְפַּזְרוּ עַל גַּבֵּי קַרְקַע, מִתָּר לְאָסְפָם לְתוֹךְ כָּלִי, וְאֵין בָּזֶה מִשּׁוּם מְעַמֵר, כֵּיוָן שֶׁאֵין זֶה מְקוֹם גִּדּוּלָם, אֲבָל אָסוּר לְהַשְׁחִילָם לְחוּט.

סִימָן סו
(ה) מְלֶאכֶת דְּשׁ
מַהוּתָהּ: פֵרוּק וְהַפְרָדַת גִּדּוּלֵי קַרְקַע מֵהַפָּסֹלֶת הַדְּבוּקָה בָהֶם אוֹ הָעוֹטֶפֶת אוֹתָם.
שִׁמּוּשָׁהּ בַּמִּשְׁכָּן: דָּשׁוּ אֶת הַפַּמְמָנִים.
א. הַדָּשׁ בְּשַׁבָּת תְּבוּאָה (שֶׁדּוֹחֵק וְלוֹחֵץ בַּקְלִפָּה הַמְכַסָה אֶת גַּרְעִינֵי הַתְּבוּאָה, וְעַל־יְדֵי כָךְ נִשְׁבָּרֶת הַקְלִפָה וְנִגְלִים גַרְעִינֵי הַתְּבוּאָה) עַל־יְדֵי בְהֵמוֹת אוֹ מַקְלוֹת אוֹ בִּמְכוֹנָה הַמְיִחֶדֶת לְדִישָׁה אָסוּר מִן הַתּוֹרָה. וְהַדָּשׁ תְּבוּאָה בְּיָדָיו – אָסוּר מִדִּבְרֵי סוֹפְרִים.
לְפִיכָךְ אָסוּר לִמְלַל מְלִילוֹת (חָטִים לַחִים) בְּשַׁבָּת בְּכַפּוֹת יְדָיו כְּדַרְכּוֹ, אֶלָּא רַק בְּרָאשֵׁי אֶצְבְּעוֹתָיו.
ב. מְתָּר לְפָרֶק שַׁרְבִיטִין שֶׁל קִטְנִיּוֹת מְבְשָׁלִים, כְּגוֹן פּוּלִין וּשְׁעוּעִית יִרְקָה, וּלְהוֹצִיא הַקְטְנִיּוֹת מֵהֶן כְּדֵי לְאָכְלָן לְאַלְתָּר וּמִיָּדן.
ג. מְתָּר לִקְלֶף בְּשַׁבָּת בָּטְנִים מִקְלִפָּתָם הַחִיצוֹנָה, וְכֵן מְתָּר לְקַלֵּף פְסְטוּק חַלְבִּי וְכַיּוֹצֵא בָזֶה בִּשְׁעַת אֲכִילָתָם, וְאֵין בָּזֶה אָסוּר מִשּׁוּם דָּשׁ, כְּשֵׁם שֶׁמִּתָּר לְשַׁבֵּר קְלִפַּת שְׁקֵדִים וֶאֱגוֹזִים כְּדֵי לִטּל הָאֹכֶל שֶׁבְּתוֹכָם.
דִין סְחִיטָה בְּפֵרוֹת
ה. הַסּוֹחֵט בְּשַׁבָּת זֵיתִים לַעֲשׂוֹת שֶׁמֶן, אוֹ עֲנָבִים לַעֲשׂוֹת יַיִן חַיָּב מִן הַתּוֹרָה מִשּׁוּם מְפָרֵק, שֶׁהִיא תּוֹלָדָה שֶׁל מְלֶאכֶת דָּשׁ.
ן. אָסוּר מִדִּבְרֵי סוֹפְרִים לִסְחֹט תּוּתִים אוֹ רִמּוֹנִים, שֶׁהוֹאִיל וּמִקְצָת בְּנֵי אָדָם סוֹחֲטִים אוֹתָם כְּזֵיתִים וַעֲנָבִים, לְפִיכָךְ גָּזְרוּ בָהֶם שֶׁמָּא יָבֹא לִסְחֹט גַּם זֵיתִים וַעֲנָבִים.
ז. מַשְׁקִין שֶׁזָבוּ מֵאֲלֵיהֶן בְּשַׁבָּת מַעֲנָבִים, אֲפִלּוּ הָיוּ עוֹמְדִים לַאֲכִילָה – אָסוּר לִשְׁתּוֹתָם בְּשַׁבָּת, גְּזֵרָה שֶׁמָּא יִסְחַט בַּיָּדַיִם. אֲבָל מַשְׁקִים הַיּוֹצְאִים מִתּוּתִים וְרִמּוֹנִים אִם עוֹמְדִים לְמַשְׁקִים, אָסוּר לִשְׁתּוֹת הַמַּשְׁקִים הַיּוֹצְאִים. וְאִם עוֹמְדִים לַאֲכִילָה, הַמַּשְׁקִים הַיּוֹצְאִים מְתָּרִים בְּשַׁבָּת.
ח. כָּל־פְּרִי שֶׁנּוֹהֲגִים בְּאֵיזֶה מָקוֹם לְסְחֲטוֹ כְּדֵי לִשְׁתּוֹת מֵימָיו לְצְמָא אוֹ תַעֲנוּג, אָסוּר לְסְחֲטוֹ בְּכָל־הָעוֹלָם כְּדִין תּוּתִים וְרִמּוֹנִים. לְפִיכָךְ אָסוּר לִסְחֹט אַגָּסִים וְתַפּוּחֵי עֵץ, שֶׁיֵּשׁ מְקוֹמוֹת שֶׁרְגִילִים לְסְחֲטָם. וְכֵן מַנְגּוֹ וַאֲנָנָס אָסוּר לְסְחֲטָם מִדִּבְרֵי סוֹפְרִים וְכַיּוֹצֵא בְאֵלּוּ, שֶׁהַדֶּרֶךְ לְסְחֲטָם לְמֵימֵיהֶם בְּכָל־הָעוֹלָם.
ט. פַרוֹת שֶׁאֵין דֶּרֶךְ לְסְחֲטָם לְמַשְׁקִים, כְּגוֹן אֲבַטִיחַ וְחָבוּשׁ וְכַיּוֹצֵא בָהֶם מִתָּר לְסְחֲטָם לִשְׁתּוֹת מֵימֵיהֶם. וַאֲפִלּוּ אִם הַדֶּרֶךְ לְסְחֲטָם לְצֹרֶךְ רְפוּאָה מֶתָּר. וְדַוְקָא בְּיָדוֹ, אֲבָל לֹא בְּמַסְחֵטָה יָדָנִית.
י. מִתָּר לִסְחט אֶשְׁכּוֹל עֲנָבִים בְּשַׁבָּת לְתוֹךְ קְדֵרָה שֶׁיֵּשׁ בָּהּ תַּבְשִׁיל אוֹ לְתוֹךְ קְעָרָה שֶׁיֵּשׁ בָּהּ אֶכְלִים, שֶׁמַּשְׁקֶה הַבָּא לְתוֹךְ הָאֹכֶל – כְּאֹכֶל. וְכָל־שֶׁכֵּן שֶׁמְתָּר לִסְחט שְׁאָר פַּרוֹת, כְּגוֹן תַּפּוּחִים וְכַיּוֹצֵא בָזֶה, לְתוֹךְ אֹכֶל. לְפִיכָךְ מְתָּר לִסְחט תַּפּוּזִים לְתוֹךְ סָלַט פּרוֹת כְּדֵי לְהַטְעִימוֹ, וּבִלְבַד שֶׁיִּהְיֶה רֹב הַמַּשְׁקָה נִבְלֶע בָּאֹכֶל.
יאן. מְתָּר לִסְחֹט לִימוֹנִים בְּשַׁבָּת אֲפִלּוּ לְמַשְׁקָה, כֵּיוָן שֶׁאֵין הַדֶּרֶךְ לְסְחֲטָם כְּדֵי לִשְׁתּוֹתָם כְּמוֹת שֶׁהֵם.
יב. מֵעִקַר הַדִּין מְתָּר לִמְצֹץ בְּשַׁבָּת בְּפִיו הַמִּיץ שֶׁבְּתוֹךְ הָעֲנָבִים אוֹ בִּשְׁאָר פּרוֹת הָעוֹמְדִים לְמַשְׁקִים, שֶׁאֵין דֶּרֶךְ סְחִיטָה בְּכָךְ. וְהַמַּחְמִיר שָׁלֹא לִמְצֹץ עֲנָבִים (שֶׁסְחִיטְתָם אֲסוּרָה מִן הַתּוֹרָה), תָּבֹא עָלָיו בְּרָכָה. אֲבָל שְׁאָר פּרוֹת, כְּגוֹן תּוּתִים וְרִמּוֹנִים וְכַיּוֹצֵא בָהֶם, שֶׁסְחִיטְתָם אֲסוּרָה מִדִּבְרֵי סוֹפְרִים מְתָּר. וְדַוְקָא כְּשֶׁמּוֹצֵץ הָעֲנָבִים דֶּרֶךְ יְנִיקָה וְאֵינוֹ מַכְנִיסָם לְפִיו, אֲבָל אִם מַכְנִיסָם לְפִיו וּמוֹצֵץ הַמַּשְׁקָה וּמַשְׁלִיךְ הַחַרְצָנִים וְהַזַגִים דֶּרֶךְ מַאֲכָל הוּא, וּמְתָּר לְכָלֵי עָלְמָא.
דִין חֲלִיבָה בְּשַׁבָּת
ין. הַחוֹלֵב אֶת הַבְּהֵמָה בְּשַׁבָּת חַיָּב מִשּׁוּם מְפָרֵק, שֶׁהוּא תּוֹלָדָה שֶׁל דְּשׁ. שֶׁהַמְפָרַק הֶחָלָב מֵעֲטִינֵי הַבְּהֵמָה, דּוֹמֶה לִמְפָרֵק גַרְעִינֵי תְבוּאָה מִתּוֹךְ הַשְּׁבֹּלֶת שֶׁלָּהֶם.
יד. אָסוּר לַחֲלֹב בְּהֵמָה בְּשַׁבָּת אֲפִלּוּ לְתוֹךְ כְּלִי שֶׁיֵּשׁ בּוֹ אֹכֶל וְהֶחָלָב בָּא לְצֹרֶךְ הָאֹכֶל לְתַקְנוֹ אוֹ שֶׁנִּבְלֶע כָּלּוֹ בָּאֹכֶל.
טו. הַחֲלִיבָה הַמֶּתֶּרֶת הִיא:
א. שֶׁיֹּאמַר לְנָכְרִי לַחֲלֹם פָּרוֹתָיו. שֶׁאֲמִירָה לְגוֹי בִּמְקוֹם צַעַר בַּעֲלֵי חַיִּים הֶתָּרָה. (וְהֶחָלָב אָסוּר לְבוֹ בַּיּוֹם).
ב. בְּמָקוֹם שֶׁאֵין מָצוּי נָכְרִי, מְתָּר לְיִשְׂרָאֵל לַחֲלֹב פָּרוֹתָיו בִּתְנַאי שֶׁיַּחֲלֹב עַל גַּבֵּי הַקַרְקַע אוֹ לְתוֹךְ כְּלִי שֶׁיֵּשׁ בְּתוֹכוֹ נַפְטְ, שֶׁיֵּלֵךְ יָשָׁר לְאָאִבּוּד, אֲבָל לֹא לְתוֹךְ כְּלִי רֵיק. וְאִם חוֹלֵב עַל־יְדֵי צִנּוֹר חַשְׁמַלִי הַפּוֹעַל עַל־יְדֵי שָׁעוֹן, שֶׁמַלְבִּישׁ הַצְנּוֹר עַל עֲטִינֵי הַפָּרָה וּלְאַחַר זְמַן נִשְׁאָב הֶחָלָב מֵעַצְמוֹ - מְתָּר אַף לְתוֹךְ כְּלִי רֵיק, כֵּיוָן שֶׁרַק גּוֹרֵם לַחֲלִיבָה, וְאֵינוֹ חוֹלֵב בְּצוּרָה יְשִׁירָה. וּבִלְבַד שֶׁטְפּוֹת הָרִאשׁוֹנוֹת שֶׁל הֶחָלָב הוֹלְכוֹת לְאִבּוּד.

סִימָן סז
(ו) מְלֶאכֶת זוֹרֶה
מַהוּתָהּ: הַפְרָדַת הַמּוֹץ, דְּהַיְנוּ הַקְלִפּוֹת, מִגַּרְעִינֵי הַתְּבוּאָה בְּאֶמְצָעוּת כָּלִי בְּשִׁתּוּף כֹּחַ הָרוּחַ.
שָׁמוּשָׁהּ בַּמִּשְׁכָּן: זָרוּ הַסַמְמָנִים בְּרַחַת (כְּלִי "אֵת" גָּדוֹל) לָרוּחַ.
מַלְאֲכוֹת הַיּוֹרֶה, הַבּוֹרֵר וְהַמְרַקֵד (נבארן לקמן), אַף־עַל־פִּי שֶׁנִּרְאוֹת דּוֹמוֹת זוֹ לְזוֹ, שָׁמַהוּת מְלָאכוֹת אֵלּוּ הִיא הַפְרָדַת הַפְסֹלֶת הַמְעֹרֶבֶת בָּאֹכֶל, מִכָּל־מָקוֹם נֶחְשָׁבוֹת לִשְׁלֹשָׁה אָבוֹת, הוֹאִיל וְכַלָּן הָיוּ בַמִּשְׁכָּן.
א. הַיּוֹרֶה בְּשַׁבָּת תְּבוּאָה אוֹ קִטְנִית וְכַיּוֹצֵא בָהֶם, בְּרַחַת (כְּלִי "אֵת" גָּדוֹל) כְּנֶגֶד הָרוּחַ, כְּדֵי שֶׁהָרוּחַ יַפְרִיחַ אֶת הַמּוֹץ (הַקְלִפּוֹת) מֵהֶם, שֶׁהֵם קַלִים וְהַתְּבוּאָה נִשְׁאֶרֶת הֲרֵי זֶה אָסוּר מִן הַתּוֹרָה מִשּׁוּם אַב מְלָאכָה שֶׁל זוֹרֶה.
ב. הַלּוֹקֵחַ בְּיָדָיו תְּבוּאָה אוֹ קִטְנִית, אָסוּר לִנְפֹּחַ בְּפִיו כְּדֵי לְהַפְרִיחַ הַמּוֹץ מֵהֶם. וְכֵן הַדִּין לֶאֱמֹר אִם לוֹקְחָן בִּשְׁתִּי יָדָיו וְזוֹרְקָן לְרוּחַ, אוֹ שֶׁמַּעֲבִירְן מִיָּד לְיָד עַד שֶׁיִּפְּלוּ הַקְלִפּוֹת. אֲבָל מְתָּר לִזְרֹק בְּיָדוֹ הָאַחַת כְּלַפֵי מַעְלָה וְשׁוּב לְתָפְסָן. לְפִיכָךְ, בָּטְנִים אוֹ שְׁקֵדִים הַמְעֹרָבִים בִּקְלִפָּתָם וְרוֹצֶה לְהָסִיר הַקְלִפּוֹת עַל־יְדֵי רוּחַ, מִתָּר לִזְרֹק בְּיָדוֹ הָאַחַת כְּלַפֵי מַעְלָה.
ג. אֵין אָסוּר לִירֹק בְּשַׁבָּת, אַף שֶׁהָרוּחַ מְפַזְרַתּוּ, שֶׁכֵּן אֵינוֹ מַפְרִיד שׁוּם פְּסֹלֶת מֵהָרֹק. וְלָכֵן מִתָּר לְהִשְׁתַּמֵּשׁ בְּשַׁבָּת בִּמְטַהֵר אֲוִיר אַף שֶׁבַּלְחִיצָה הַבֹּשֶׁם מִתְפַּזֵר לִרְסִיסִים דַּקִים מְאֹד.

סִימָן סח
(ז) מְלֶאכֶת בּוֹרֵר
מַהוּתָהּ: הַפְרָדָה לְסוּגִים שׁוֹנִים אוֹ הַפְרָדַת פְּסֹלֶת מֵאֹכֶל.
שְׁמוּשָׁהּ בַּמִּשְׁכָּן: אַחַר זְרִיָּת הַמּוֹץ בָּרוּחַ, שֶׁהֵן הַפְּסֹלֶת הַקַלָּה, בָּרְרוּ הַפָּסֹלֶת הַכְּבֵדָה, דְּהַיְנוּ הַצְרוֹרוֹת וְהֶעָפָר הַמְעֹרָבִים בַּגַּרְעִינִים.
א. הָיוּ לְפָנָיו אֹכֶל וּפְסֹלֶת מְעֹרָבִים אָסוּר מִן הַתּוֹרָה לִבְרֹר (כְּלוֹמַר לְהָסִיר) אֶת הַפְּסֹלֶת מִן הָאֹכֶל בְּכָל־עִנְיָן, אֲפִלּוּ אִם עוֹשֶׂה כֵּן עַל דַּעַת לֶאֱכֹל מִיָּד, וּבְיָדוֹ, וְאַף אִם בּוֹרֵר רַק חֵלֶק מֵהַפְסֹלֶת. אֲבָל מְתָּר לִבְרֹר אֹכֶל מִתּוֹךְ הַפְסֹלֶת (בַּדְּרָכִים שֶׁיְבֹאֲרוּ לְקַמָּן).
ב. הַבּוֹרֵר בִּכְלִי שֶׁהַדֶּרֶךְ לִבְרֹר בּוֹ, כְּגוֹן נָפָה אוֹ מִסְנֶנֶת, אֲפִלּוּ בּוֹרֵר אֹכֶל מִתּוֹךְ פְּסֹלֶת עַל דַּעַת לֶאֱכֹל מִיָּד – חַיָּב. וְאִם בֵּרֵר בִּכְלִי שֶׁאֵינוֹ מְיִחָד לִבְרֵרָה – אָסוּר מִדִּבְרֵי סוֹפְרִים.
ג. הַבּוֹרֵר בְּיָדוֹ אֹכֶל מִתּוֹךְ פְּסֹלֶת עַל דַּעַת לְהָנִיחַ, דְּהַיְנוּ לֶאֱכֹל לְאַחַר שָׁעָה חַיָּב.
ד. הַבְּרֵרָה הַמַּתֶּרֶת הִיא בִּתְנַאי:
א. שֶׁיִּבְרֹר אֹכֶל מִתּוֹךְ פְּסֹלֶת.
ב. שֶׁיִּבְרֹר עַל דַּעַת לֶאֱכֹל מִיָּד.
ג. שֶׁיִּבְרֹר בַּיָּד וְלֹא בַּכָּלִי.
ה. הַבּוֹרֵר אֹכֶל מִתּוֹךְ פָּסֹלֶת כְּדֵי לֶאֱכֹל מִיָּד, צָרִיךְ לִבְרֹר סָמוּךְ לַסְעוּדָה מַמָּשׁ. וְלָכֵן אֵין לְאַנְשֵׁי הַבַּיִת לִבְרֹר אֶלָּא לְאַחַר צֵאתָם שֶׁל בַּעַל הַבַּיִת וְהַנִּלְוִים אֵלָיו מִבֵּית הַכְּנֶסֶת. וּמְתָּר לִבְרֹר כָּל־מַה־שֶׁיִצְטָרֵךְ לֶאֱכֹל בְּאוֹתָהּ סְעוּדָה אַף שֶׁמַּאֲרִיךְ בָּהּ כַּמָּה שָׁעוֹת.
ן. הַבּוֹרֵר אֹכֶל מִתּוֹךְ פְּסֹלֶת לֶאֱכֹל מִיָּד, אֲפִלּוּ עַל־יְדֵי כַּף אוֹ מַזְלֵג מַתָּר, וְאֵין זֶה נֶחְשָׁב כְּבוֹרֵר בְּכָלִי, הוֹאִיל וְדֶרֶךְ בְּנֵי אָדָם לֶאֱכֹל עַל־יְדֵי כַּף וּמַזְלֵג, וְנֶחְשָׁב הָעוֹסֵק בַּאֲכִילָה כְּבוֹרֵר בְּיָדוֹ.
ז. הֵבִיאוּ לְפָנָיו בְּשַׁבָּת שְׁנֵי מִינֵי אֱכָלִים מְעֹרָבִים, כְּגוֹן אֹרֶז וַאֲפוּנָה הַמְעֹרָבִים, אוֹ גַּרְעִינִים וּבָטְנִים הַמְעֹרָבִים, וְהוּא חָפֵץ בְּאֶחָד מֵהֶם אוֹתוֹ מַאֲכָל שֶׁחָפֵץ בּוֹ עַתָּה נֶחְשָׁב לְאֹכֶל, וְהַמַּאֲכָל הַשֵּׁנִי שֶׁאֵינוֹ חָפֵץ בּוֹ עַתָּה נֶחְשָׁב לִפְסֹלֶת. וְלָכֵן יֵשׁ לוֹ לִבְרֹר הָאֹכֶל שֶׁרוֹצֶה בּוֹ עַתָּה עַל דַּעַת לְאָכְלוֹ מִיָּד.
ה. הָיוּ לְפָנָיו מֵאוֹתוֹ מִין, אֶלָּא שֶׁיֵּשׁ בּוֹ גְדוֹלִים וּקְטַנִים - אֵין בָּזֶה דִין בּוֹרֵר, וְיָכוֹל לִבְרֹר אַף אוֹתוֹ שֶׁרוֹצֶה לְהָנִיחַ לְאַחַר זְמַן, אֲבָל שְׁנֵי מִינֵי דָגִים מְעֹרָבִים, וְכֵן חֲתִיכוֹת בָּשָׂר שֶׁחֶלְקָן מְטַגָּנוֹת וְחֶלְקָן צְלוּיוֹת, וְכֵן חֶלְקֵי עוֹפוֹת שׁוֹנִים - נֶחְשָׁבִים כִּשְׁנֵי מִינֵי אֱכָלִים, וְצָרִיךְ לִבְרֹר אוֹתוֹ שֶׁרוֹצֶה לְאָכְלוֹ מִיָּד.
ט. מִתָּר לִבְרֹר בְּשַׁבָּת הָעֲצָמוֹת הַדַּקִים שֶׁבְּתוֹךְ הַדָּגִים, שֶׁכֵּן דֶּרֶךְ אֲכִילָתוֹ, וְאֵינוֹ נֶחְשָׁב כְּבוֹרֵר פָּסֹלֶת מִתּוֹךְ אֹכֶל. אֲבָל עֲצָמוֹת שֶׁנִּפְרְדוּ מִן הַדָּגִים שֶׁבַּצַלַחַת, לֹא יִבְרְרֵם לְסַלְקָם מִשָּׁם.
וּ. הָאוֹכֵל תַּפּוּחַ אוֹ אַגָּס הַפָּגוּם בְּחֶלְקוֹ, מְתָּר לוֹ לַחְתּךְ בְּשַׁבָּת בְּסַכִּין אֶת הַחֵלֶק הַפָּגוּם כְּדֵי לֶאֱכֹל אֶת הַנִּשְׁאָר, וְאֵין זֶה נֶחְשָׁב כְּבוֹרֵר פְּסֹלֶת מִתּוֹךְ אֹכֶל.
יא. הָאוֹכֵל תְּמָרִים אוֹ מִשְׁמְשִׁים, מְתָּר לְהוֹצִיא מֵהֶם הַגַּרְעִין בַּיָּד בִּשְׁעַת אֲכִילָתוֹ.
יב. שׁוּם וּבָצֵל וְרִמּוֹן וּבֵיצִים, מְתָּר לְקָלְפָם עַל מְנֶת לְאָכְלָם מִיָּד. אֲבָל אִם רוֹצֶה לְאָכְלָם לְאַחַר שָׁעָה אָסוּר, כְּדִין בּוֹרֵר אֹכֶל מִתּוֹךְ פְּסֹלֶת. אֲבָל אַגָּסִים וְתַפּוּחִים וּמְלָפְפוֹן, שֶׁהַקְלִפָּה דְּבוּקָה בָּהֶם מְאֹד, וּכְשֶׁקּוֹלְפִים אוֹתָם בְּסַכִּין נִקְלֶף גַּם חֵלֶק מֵהַפְּרִי לֹא שַׁיָּךְ בָּזֶה בּוֹרֵר כְּלָל, וְלָכֵן מִתָּר לְקָלְפָם אַף עַל דַּעַת לְאָכְלָם לְאַחַר זְמַן.
ין. מְתָּר לְהָדִיחַ פַּרוֹת וִירָקוֹת בְּמַיִם כְּדֵי לְהָסִיר הַלְכְלוּךְ שֶׁעֲלֵיהֶם עַל מְנָת לְאָכְלָם מִיָּד, אֲבָל אָסוּר לִשְׁרוֹתָם בִּכְלִי מָלֵא מַיִם לְהוֹרִיד הֶעָפָר וְהַלְכְלוּךְ מֵהֶם אַף שֶׁרוֹצֶה לְאָכְלָם מִיָּד.
יד. פְּסֹלֶת וְאֹכֶל הַמֶּנָּחִים זֶה לְצַד זֶה, זֶה מִשְׂמֹאל וְזֶה מִיָּמִין, אֲבָל אֵינָם מְעֹרָבִים יַחַד לֹא שַׁיָּךְ בָּהֶם אָסוּר בּוֹרֵר, וְיָכוֹל לִטּל אֵיזֶה מֵהֶם שֶׁיִּרְצָה אֲפִלּוּ דַּעְתּוֹ לֶאֱכֹל לְאַחַר זְמַן.
טן. הַבּוֹרֵר אֹכֶל מִתּוֹךְ פְּסֹלֶת לְצֹרֶךְ אוֹרְחִים, לֹא יִבְרֹר לָהֶם קְעָרָה מְלֵאָה כְּשֶׁיּוֹדֵעַ שֶׁהֵם לֹא יִטְעֲמוּ אֶלָּא קְצָת, וְנִמְצָא שֶׁמַה שֶׁנִּשְׁאָר בַּקְעָרָה נִבְרַר שָׁלֹא לְאַלְתַּר, אֶלָּא עָלָיו לְשַׁעֵר מַה־שֶׁרָאוּי לָאוֹרְחִים וְלֹא יוֹתֵר.
טז. נָפַל זְבוּב אוֹ יַתּוּשׁ לְתוֹךְ מַשְׁקֶה מֵעִקַר הַדִּין מְתָּר לְהוֹצִיאָם בְּשַׁבָּת בַּיָּד אוֹ בְּכַף אַף בְּלִי מַשְׁקָה כְּלָל, שֶׁאֵין בְּרֵרָה בְּלַח. וּמִכָּל־מָקוֹם, נָכוֹן לְהַחְמִיר וְלַהֲסִירָם יַחַד עִם מַשְׁקֶה, אוֹ יַטָּה הַכּוֹס לַצַד וְיִשָּׁפֵךְ מְעַט מַשְׁקָה עִם הַזְבוּב.
ין. כְּשֵׁם שֶׁיֵּשׁ אָסוּר בְּרֵרָה בִּשְׁנֵי מִינֵי אֱכָלִין כָּךְ יֵשׁ אָסוּר בְּרֵרָה בְּכֵלִים, בְּגָדִים אוֹ סְפָרִים הַמְעֹרָבִים זֶה בָזֶה. וְלָכֵן כְּשֶׁרוֹצֶה לִבְרֹר כְּלִי אוֹ בֶּגֶד מֵחֲבֵרוֹ, יִבְרֹר הַחֵפֶץ שֶׁנִּצְרָךְ אֵלָיו וְיִשְׁתַּמֵשׁ בּוֹ לְאַלְתַּר. וּמִכָּל־מָקוֹם, מְתָּר לְהָסִיר בְּשַׁבָּת בְּגָדִים מֵעַל קוֹלָב אַף שֶׁאֵינוֹ צָרִיךְ לָהֶם, אֶלָּא כַּוּנָתוֹ לְהוֹצִיא אֶת הַבֶּגֶד הַפְּנִימִי שֶׁנִּמְצָא תַּחְתֵּיהֶם, שֶׁהֲרֵי אֵינוֹ יָכוֹל לְהַגִּיעַ לַבֶּגֶד שֶׁרוֹצֶה מִבְּלִי שֶׁיָּסִיר הַבְּגָדִים שֶׁעָלָיו. וְכֵן מְתָּר לְסַדֵר הַחְמָּשִׁים וְהַמִּדּוּרִים שֶׁהִנִּיחוּ הַקָּהָל בְּבֵית־הַכְּנֶסֶת, וּלְמַיְנָם בִּמְקוֹמָם.
יח. סַכִּינִים, כַּפּוֹת וּמַזְלְגוֹת שֶׁמְעֹרָבִים יַחַד, מְתָּר לִבְוֹר לִתְּנָם כָּל־אֶחָד בִּמְקוֹמוֹ לְצֹרֶךְ הַסְעוּדָה שֶׁלְאַחַר מִכֵּן, מִפְּנֵי שֶׁאֵין כָּאן פְּסֹלֶת וְאֹכֶל הַמְעֹרָבִים, שֶׁהֲרֵי כְּלָּם שָׁוִים בְּעֵינָיו.
יט. מְחַבֵּץ וּמְגַבֵּן הֵן תּוֹלֶדֶת בּוֹרֵר, וְהַיְנוּ שֶׁעוֹשֶׂה חֶמְאָה (מְחַבֵּץ), וְכֵן לַעֲשׂוֹת גְּבִינָה אֲפִלּוּ רַכָּה (מְגַכֵּן), שֶׁבָּזֶה בּוֹרֵר אֶת הַשָּׁמָּן מִן הַמַּיִם שֶׁבֶּחָלָב.
